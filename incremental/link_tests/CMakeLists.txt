cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_playable_link_check LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The final PE linkage oracle requires MSVC x86: configure with -A Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(TH20_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(th20_playable_link_check link_check.cpp "${TH20_ROOT}/src/binary.cpp" "${TH20_ROOT}/src/pe.cpp")
target_include_directories(th20_playable_link_check PRIVATE "${TH20_ROOT}/include" "${TH20_ROOT}/native_recovered")
target_compile_features(th20_playable_link_check PRIVATE cxx_std_17)
target_compile_options(th20_playable_link_check PRIVATE /W4 /fp:strict /arch:SSE2 /utf-8 /Oy-)
target_link_options(th20_playable_link_check PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
target_link_libraries(th20_playable_link_check PRIVATE bcrypt)
