cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_native_bridge LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "This bridge requires MSVC x86: configure with -A Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(TH20_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(th20_native_bridge SHARED bridge.cpp bridge.def)
target_include_directories(th20_native_bridge PRIVATE "${TH20_ROOT}/native_recovered")
target_compile_features(th20_native_bridge PRIVATE cxx_std_17)
target_compile_options(th20_native_bridge PRIVATE /W4 /fp:strict /arch:SSE2 /utf-8)
target_link_options(th20_native_bridge PRIVATE /DYNAMICBASE /NXCOMPAT)

add_executable(th20_bridge_abi_check abi_check.cpp "${TH20_ROOT}/src/binary.cpp" "${TH20_ROOT}/src/pe.cpp")
target_include_directories(th20_bridge_abi_check PRIVATE "${TH20_ROOT}/native_recovered" "${TH20_ROOT}/include")
target_compile_features(th20_bridge_abi_check PRIVATE cxx_std_17)
target_compile_options(th20_bridge_abi_check PRIVATE /W4 /fp:strict /arch:SSE2 /utf-8)
target_link_options(th20_bridge_abi_check PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
target_link_libraries(th20_bridge_abi_check PRIVATE bcrypt)
file(SHA256 "${TH20_ROOT}/native_recovered/native_core.hpp" NATIVE_CORE_SHA)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${TH20_ROOT}/native_recovered/native_core.hpp")
target_compile_definitions(th20_bridge_abi_check PRIVATE TH20_NATIVE_CORE_SHA256="${NATIVE_CORE_SHA}")
