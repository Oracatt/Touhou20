cmake_minimum_required(VERSION 3.16)
project(th20_native_cpu_compare LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The hardware oracle requires 32-bit MSVC: configure with -A Win32")
endif()
add_executable(th20_native_cpu_compare ../native_cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
target_include_directories(th20_native_cpu_compare PRIVATE ../../include ../../native_recovered)
target_compile_features(th20_native_cpu_compare PRIVATE cxx_std_17)
target_compile_options(th20_native_cpu_compare PRIVATE /W4 /fp:strict /arch:SSE2 /utf-8)
target_link_options(th20_native_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO)
target_link_libraries(th20_native_cpu_compare PRIVATE bcrypt)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../native_recovered/native_core.hpp" NATIVE_CORE_SHA)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../native_recovered/native_core.hpp")
target_compile_definitions(th20_native_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="${NATIVE_CORE_SHA}")
