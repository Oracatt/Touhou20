cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(th20_native_recovered LANGUAGES CXX)
endif()
add_library(th20_native_recovered SHARED native_exports.cpp)
target_compile_features(th20_native_recovered PUBLIC cxx_std_17)
target_include_directories(th20_native_recovered PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_compile_options(th20_native_recovered PRIVATE /W4 /fp:strict)
else()
    target_compile_options(th20_native_recovered PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
endif()
