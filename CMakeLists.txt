cmake_minimum_required(VERSION 3.16)
project(Touhou20Reconstruction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(th20_binary STATIC src/binary.cpp src/ecl.cpp src/pe.cpp)
target_include_directories(th20_binary PUBLIC include)
if(MSVC)
  target_compile_options(th20_binary PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(th20_binary PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_subdirectory(native_recovered)
add_executable(th20_inspect src/main.cpp)
target_link_libraries(th20_inspect PRIVATE th20_binary)

include(CTest)
if(BUILD_TESTING)
  add_executable(th20_binary_tests tests/binary_tests.cpp)
  target_link_libraries(th20_binary_tests PRIVATE th20_binary)
  add_test(NAME th20_binary_tests COMMAND th20_binary_tests)
endif()
