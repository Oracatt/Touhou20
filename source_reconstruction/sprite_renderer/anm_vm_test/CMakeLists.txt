cmake_minimum_required(VERSION 3.20)
project(th20_anm_vm_test LANGUAGES CXX)
add_executable(th20_anm_vm_cpu_compare ../anm_vm_cpu_compare.cpp ../anm_vm.cpp ../animation.cpp ../../ecl_vm/math.cpp ../../../src/binary.cpp ../../../src/pe.cpp)
target_include_directories(th20_anm_vm_cpu_compare PRIVATE ../../../native_recovered ../../../include)
target_compile_options(th20_anm_vm_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_compile_features(th20_anm_vm_cpu_compare PRIVATE cxx_std_17)
target_link_libraries(th20_anm_vm_cpu_compare PRIVATE bcrypt)
target_link_options(th20_anm_vm_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../anm_vm.cpp" ANM_CPP_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../anm_vm.hpp" ANM_HPP_SHA)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../anm_vm.cpp ../anm_vm.hpp)
target_compile_definitions(th20_anm_vm_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused" TH20_ANM_VM_CPP_SHA="${ANM_CPP_SHA}" TH20_ANM_VM_HPP_SHA="${ANM_HPP_SHA}")
function(bind_source_hash name path)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" source_sha)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  target_compile_definitions(th20_anm_vm_cpu_compare PRIVATE "${name}=\"${source_sha}\"")
endfunction()
bind_source_hash(TH20_ANM_ANIMATION_CPP_SHA ../animation.cpp)
bind_source_hash(TH20_ANM_ANIMATION_HPP_SHA ../animation.hpp)
bind_source_hash(TH20_ANM_MATH_CPP_SHA ../../ecl_vm/math.cpp)
bind_source_hash(TH20_ANM_MATH_HPP_SHA ../../ecl_vm/math.hpp)
bind_source_hash(TH20_ANM_CORE_SHA ../../../native_recovered/native_core.hpp)
bind_source_hash(TH20_ANM_ORACLE_SHA ../anm_vm_cpu_compare.cpp)
