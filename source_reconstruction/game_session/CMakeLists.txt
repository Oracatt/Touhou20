cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_game_session LANGUAGES CXX)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(th20_game_session STATIC session.cpp)
target_compile_options(th20_game_session PRIVATE /W4 /utf-8)
target_compile_features(th20_game_session PUBLIC cxx_std_20)
if(TH20_BUILD_ORACLE)
  add_executable(th20_session_cpu_compare cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
  target_link_libraries(th20_session_cpu_compare PRIVATE th20_game_session bcrypt)
  target_include_directories(th20_session_cpu_compare PRIVATE ../../include ../../native_recovered)
  target_compile_features(th20_session_cpu_compare PRIVATE cxx_std_20)
  target_compile_options(th20_session_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
  target_link_options(th20_session_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO)
  target_compile_definitions(th20_session_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused")
endif()
