cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_trophy_oracle LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(th20_trophy_cpu_compare compare.cpp ../codec.cpp ../queue.cpp ../../runtime_core/runtime_core.cpp ../../core_scheduler/scheduler.cpp ../../../src/binary.cpp ../../../src/pe.cpp)
target_compile_features(th20_trophy_cpu_compare PRIVATE cxx_std_20)
target_include_directories(th20_trophy_cpu_compare PRIVATE ../../../native_recovered ../../../include ../../core_scheduler)
target_compile_options(th20_trophy_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2 /Gy)
target_link_options(th20_trophy_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /OPT:REF /STACK:8388608)
target_link_libraries(th20_trophy_cpu_compare PRIVATE bcrypt user32)
target_compile_definitions(th20_trophy_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused")
set(HASH_INPUTS ../trophy.hpp ../codec.cpp ../queue.cpp ../../runtime_core/runtime_core.cpp ../../runtime_core/runtime_core.hpp)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.json" "{\n")
set(HASH_SEPARATOR "")
foreach(INPUT IN LISTS HASH_INPUTS)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${INPUT}" INPUT_HASH)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.json" "${HASH_SEPARATOR}  \"${INPUT}\":\"${INPUT_HASH}\"")
  set(HASH_SEPARATOR ",\n")
endforeach()
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.json" "\n}\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${HASH_INPUTS})
