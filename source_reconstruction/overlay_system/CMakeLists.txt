cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_overlay_system LANGUAGES CXX)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(th20_overlay_system STATIC weapon.cpp basic_weapons.cpp bar_weapon.cpp ring_weapon.cpp shield_weapon.cpp cloud_weapon.cpp yellow_weapon.cpp weapon_cancellation.cpp owner.cpp queries.cpp factory.cpp lifecycle.cpp frame.cpp visuals.cpp environment.cpp entry_adapter.cpp)
target_compile_features(th20_overlay_system PUBLIC cxx_std_20)
target_compile_options(th20_overlay_system PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_include_directories(th20_overlay_system PRIVATE ../../native_recovered ../../include ../core_scheduler)
target_link_libraries(th20_overlay_system PUBLIC th20_runtime_state th20_core_scheduler th20_player_entity th20_game_session th20_sprite_renderer)
