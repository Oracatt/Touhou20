cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_audio_runtime LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC -A Win32 for recovered DirectSound layouts")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT TARGET th20_runtime_core)
  add_subdirectory(../runtime_core runtime_core_dependencies)
endif()
add_library(th20_audio_runtime STATIC audio_state.cpp audio_constants.cpp direct_sound.cpp wave_reader.cpp music_stream.cpp sound_controller.cpp)
target_include_directories(th20_audio_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(th20_audio_runtime PUBLIC cxx_std_20)
target_compile_options(th20_audio_runtime PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_link_libraries(th20_audio_runtime PUBLIC th20_runtime_core dsound dxguid user32 winmm)
add_library(th20_audio_entry_adapter STATIC entry_adapter.cpp)
target_link_libraries(th20_audio_entry_adapter PUBLIC th20_audio_runtime)
target_compile_options(th20_audio_entry_adapter PRIVATE /W4 /utf-8)
if(TARGET th20_resource_manager)
  target_link_libraries(th20_audio_entry_adapter PUBLIC th20_resource_manager th20_platform_services)
endif()
add_executable(th20_audio_backend_smoke backend_smoke.cpp)
target_link_libraries(th20_audio_backend_smoke PRIVATE th20_audio_runtime)
target_compile_options(th20_audio_backend_smoke PRIVATE /W4 /utf-8)
option(TH20_BUILD_AUDIO_ORACLE "Build isolated original CPU audio validation" ON)
if(TH20_BUILD_AUDIO_ORACLE)
  add_executable(th20_audio_cpu_compare cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
  target_include_directories(th20_audio_cpu_compare PRIVATE ../../native_recovered ../../include)
  target_link_libraries(th20_audio_cpu_compare PRIVATE th20_audio_runtime bcrypt)
  target_compile_options(th20_audio_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
  target_link_options(th20_audio_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
  target_compile_definitions(th20_audio_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="not-used-by-audio")
  set(AUDIO_HASH_HEADER "struct AudioSourceHash { const char* name;const char* hash; };\nstatic const AudioSourceHash audio_source_hashes[]={\n")
  foreach(source audio.hpp audio_state.cpp audio_constants.cpp direct_sound.cpp wave_reader.hpp wave_reader.cpp music_stream.hpp music_stream.cpp sound_controller.cpp entry_adapter.cpp cpu_compare.cpp test_buffer.hpp)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
    string(APPEND AUDIO_HASH_HEADER "{\"${source}\",\"${digest}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${source})
  endforeach()
  string(APPEND AUDIO_HASH_HEADER "};\n")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/audio_source_hashes.hpp" CONTENT "${AUDIO_HASH_HEADER}")
  target_include_directories(th20_audio_cpu_compare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
endif()
