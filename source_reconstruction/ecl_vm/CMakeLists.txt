cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(th20_ecl_vm LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
endif()
if(NOT TARGET th20_binary)
  add_library(th20_ecl_binary STATIC ../../src/binary.cpp ../../src/ecl.cpp ../../src/pe.cpp)
  target_include_directories(th20_ecl_binary PUBLIC ../../include)
  set(TH20_ECL_BINARY_TARGET th20_ecl_binary)
else()
  set(TH20_ECL_BINARY_TARGET th20_binary)
endif()
add_library(th20_ecl_vm STATIC vm.cpp math.cpp)
target_include_directories(th20_ecl_vm PUBLIC . ../../native_recovered)
target_link_libraries(th20_ecl_vm PUBLIC ${TH20_ECL_BINARY_TARGET})
if(MSVC)
  target_compile_options(th20_ecl_vm PRIVATE /W4 /permissive- /utf-8 /fp:strict /arch:SSE2)
endif()
if(BUILD_TESTING)
  add_executable(th20_ecl_source_tests source_tests.cpp)
  target_link_libraries(th20_ecl_source_tests PRIVATE th20_ecl_vm)
  if(MSVC)
    target_compile_options(th20_ecl_source_tests PRIVATE /utf-8 /fp:strict /arch:SSE2)
  endif()
  add_test(NAME th20_ecl_source_tests COMMAND th20_ecl_source_tests)
endif()
option(TH20_BUILD_ECL_ORACLE "Build test-only original CPU ECL oracle" OFF)
if(TH20_BUILD_ECL_ORACLE)
  add_executable(th20_ecl_cpu_compare cpu_compare.cpp)
  target_link_libraries(th20_ecl_cpu_compare PRIVATE th20_ecl_vm bcrypt)
  target_compile_options(th20_ecl_cpu_compare PRIVATE /utf-8 /arch:SSE2 /fp:strict)
  target_include_directories(th20_ecl_cpu_compare PRIVATE ../../native_recovered)
  target_compile_definitions(th20_ecl_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vm.cpp" VM_CPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vm.hpp" VM_HPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/math.cpp" MATH_CPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/math.hpp" MATH_HPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/cpu_compare.cpp" CPU_COMPARE_SHA)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS vm.cpp vm.hpp math.cpp math.hpp cpu_compare.cpp)
  target_compile_definitions(th20_ecl_cpu_compare PRIVATE TH20_ECL_CPP_SHA="${VM_CPP_SHA}" TH20_ECL_HPP_SHA="${VM_HPP_SHA}")
  target_compile_definitions(th20_ecl_cpu_compare PRIVATE TH20_ECL_MATH_CPP_SHA="${MATH_CPP_SHA}" TH20_ECL_MATH_HPP_SHA="${MATH_HPP_SHA}" TH20_ECL_CPU_COMPARE_SHA="${CPU_COMPARE_SHA}")
  target_link_options(th20_ecl_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
endif()
