cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_input LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC -A Win32 for recovered input layouts")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT TARGET th20_runtime_core)
  add_subdirectory(../runtime_core runtime_core_dependencies)
endif()
if(NOT TARGET th20_platform_services)
  add_library(th20_input_bindings STATIC ../platform_services/configuration.cpp)
  target_compile_features(th20_input_bindings PUBLIC cxx_std_17)
  set(INPUT_CONFIGURATION_TARGET th20_input_bindings)
else()
  set(INPUT_CONFIGURATION_TARGET th20_platform_services)
endif()
add_library(th20_input STATIC input_state.cpp input_win32.cpp)
target_include_directories(th20_input PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(th20_input PUBLIC cxx_std_17)
target_compile_options(th20_input PRIVATE /W4 /utf-8)
target_link_libraries(th20_input PUBLIC th20_runtime_core ${INPUT_CONFIGURATION_TARGET}
  user32 winmm xinput dinput8 dxguid ole32 oleaut32 wbemuuid)
add_library(th20_input_entry_adapter STATIC entry_adapter.cpp)
target_link_libraries(th20_input_entry_adapter PUBLIC th20_input)
target_compile_features(th20_input_entry_adapter PRIVATE cxx_std_17)
target_compile_options(th20_input_entry_adapter PRIVATE /W4 /utf-8)
include(CTest)
add_executable(th20_input_tests input_tests.cpp)
target_link_libraries(th20_input_tests PRIVATE th20_input)
target_compile_options(th20_input_tests PRIVATE /W4 /utf-8)
add_test(NAME th20_input_tests COMMAND th20_input_tests)
option(TH20_BUILD_INPUT_ORACLE "Build isolated input original CPU comparison" ON)
if(TH20_BUILD_INPUT_ORACLE)
  add_executable(th20_input_cpu_compare cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
  target_include_directories(th20_input_cpu_compare PRIVATE ../../native_recovered ../../include)
  target_compile_features(th20_input_cpu_compare PRIVATE cxx_std_17)
  target_compile_options(th20_input_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
  target_link_libraries(th20_input_cpu_compare PRIVATE th20_input bcrypt)
  target_link_options(th20_input_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
  target_compile_definitions(th20_input_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="not-used-by-input")
  foreach(source input.hpp input_state.cpp input_win32.cpp)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
    string(REPLACE "." "_" symbol "${source}")
    target_compile_definitions(th20_input_cpu_compare PRIVATE "TH20_SHA_${symbol}=\"${digest}\"")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${source})
  endforeach()
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../runtime_core/callback_owner.hpp" callback_owner_digest)
  target_compile_definitions(th20_input_cpu_compare PRIVATE "TH20_SHA_callback_owner_hpp=\"${callback_owner_digest}\"")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../runtime_core/callback_owner.hpp)
endif()
