cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_startup_scene_source LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Recovered object layout requires MSVC Win32")
endif()
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(th20_startup_scene STATIC startup.cpp shared_effects.cpp)
target_compile_features(th20_startup_scene PUBLIC cxx_std_20)
target_compile_options(th20_startup_scene PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_include_directories(th20_startup_scene PRIVATE ../../include ../../native_recovered ../core_scheduler)
target_link_libraries(th20_startup_scene PUBLIC th20_platform_window th20_sprite_renderer th20_audio_runtime th20_game_session th20_text_renderer)
target_link_libraries(th20_startup_scene PUBLIC th20_progress_state)
target_link_libraries(th20_startup_scene PUBLIC th20_effect_system)
