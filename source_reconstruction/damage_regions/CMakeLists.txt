cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_damage_regions LANGUAGES CXX)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(th20_damage_regions STATIC regions.cpp controller.cpp geometry.cpp collision.cpp damage.cpp score.cpp damage_rewards.cpp hit_callbacks.cpp entry_adapter.cpp)
target_compile_features(th20_damage_regions PUBLIC cxx_std_20)
target_compile_options(th20_damage_regions PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_include_directories(th20_damage_regions PRIVATE ../../native_recovered ../../include ../core_scheduler)
target_link_libraries(th20_damage_regions PUBLIC th20_runtime_state th20_game_session th20_core_scheduler)
