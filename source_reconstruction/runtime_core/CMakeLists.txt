cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_runtime_core LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC -A Win32 for the recovered runtime layouts")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT TARGET th20_core_scheduler)
  add_library(th20_core_scheduler STATIC ../core_scheduler/scheduler.cpp)
  target_include_directories(th20_core_scheduler PUBLIC ../core_scheduler)
  target_compile_features(th20_core_scheduler PUBLIC cxx_std_17)
endif()
add_library(th20_runtime_core STATIC runtime_core.cpp worker.cpp)
target_include_directories(th20_runtime_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(th20_runtime_core PUBLIC cxx_std_17)
target_compile_features(th20_runtime_core PRIVATE cxx_std_20)
target_compile_options(th20_runtime_core PRIVATE /W4 /utf-8)
target_link_libraries(th20_runtime_core PUBLIC th20_core_scheduler user32)
add_library(th20_runtime_entry_adapter STATIC entry_adapter.cpp)
target_link_libraries(th20_runtime_entry_adapter PUBLIC th20_runtime_core)
target_compile_features(th20_runtime_entry_adapter PRIVATE cxx_std_17)
target_compile_options(th20_runtime_entry_adapter PRIVATE /W4 /utf-8)
include(CTest)
add_executable(th20_runtime_core_tests runtime_tests.cpp)
target_link_libraries(th20_runtime_core_tests PRIVATE th20_runtime_core)
target_compile_features(th20_runtime_core_tests PRIVATE cxx_std_17)
target_compile_options(th20_runtime_core_tests PRIVATE /W4 /utf-8)
add_test(NAME th20_runtime_core_tests COMMAND th20_runtime_core_tests)
option(TH20_BUILD_RUNTIME_ORACLE "Build isolated original CPU validation target" ON)
if(TH20_BUILD_RUNTIME_ORACLE)
  add_executable(th20_runtime_cpu_compare cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
  target_include_directories(th20_runtime_cpu_compare PRIVATE ../../native_recovered ../../include)
  target_link_libraries(th20_runtime_cpu_compare PRIVATE th20_runtime_core bcrypt)
  target_compile_features(th20_runtime_cpu_compare PRIVATE cxx_std_20)
  target_compile_options(th20_runtime_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
  target_link_options(th20_runtime_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/runtime_core.cpp" RUNTIME_CPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/runtime_core.hpp" RUNTIME_HPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/worker.cpp" WORKER_CPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/worker.hpp" WORKER_HPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/worker_cpu_cases.inc" WORKER_CASES_SHA)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS runtime_core.cpp runtime_core.hpp worker.cpp worker.hpp worker_cpu_cases.inc)
  target_compile_definitions(th20_runtime_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="not-used-by-runtime"
    TH20_RUNTIME_CPP_SHA="${RUNTIME_CPP_SHA}" TH20_RUNTIME_HPP_SHA="${RUNTIME_HPP_SHA}"
    TH20_WORKER_CPP_SHA="${WORKER_CPP_SHA}" TH20_WORKER_HPP_SHA="${WORKER_HPP_SHA}" TH20_WORKER_CASES_SHA="${WORKER_CASES_SHA}")
endif()
