cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_program_entry_source LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Recovered Windows object layouts require MSVC Win32 (-A Win32)")
endif()
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT TARGET th20_core_scheduler)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../core_scheduler" "${CMAKE_CURRENT_BINARY_DIR}/core_scheduler")
endif()
add_library(th20_program_entry_source STATIC program_entry.cpp frame_schedulers.cpp winmain.cpp)
target_compile_features(th20_program_entry_source PUBLIC cxx_std_17)
target_compile_options(th20_program_entry_source PRIVATE /W4 /fp:strict /arch:SSE2 /utf-8)
target_include_directories(th20_program_entry_source PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
# This is deliberately a static source module. Missing engine functions and
# global construction are undefined symbols; there is no fake game executable.
target_link_libraries(th20_program_entry_source PUBLIC th20_core_scheduler user32 winmm)
