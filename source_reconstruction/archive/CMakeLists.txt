cmake_minimum_required(VERSION 3.16)
project(th20_source_archive LANGUAGES CXX)
add_library(th20_source_archive STATIC archive.cpp)
target_include_directories(th20_source_archive PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(th20_source_archive PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(th20_source_archive PRIVATE /W4 /permissive- /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(th20_source_archive_tests archive_tests.cpp)
  target_link_libraries(th20_source_archive_tests PRIVATE th20_source_archive)
  target_compile_features(th20_source_archive_tests PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(th20_source_archive_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_test(NAME th20_source_archive_tests COMMAND th20_source_archive_tests)
endif()
add_executable(th20_source_archive_verify verify.cpp)
target_link_libraries(th20_source_archive_verify PRIVATE th20_source_archive)
target_compile_features(th20_source_archive_verify PRIVATE cxx_std_17)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS archive.cpp archive.hpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/archive.cpp" TH20_ARCHIVE_CPP_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/archive.hpp" TH20_ARCHIVE_HPP_SHA256)
target_compile_definitions(th20_source_archive_verify PRIVATE
  TH20_ARCHIVE_CPP_SHA256="${TH20_ARCHIVE_CPP_SHA256}"
  TH20_ARCHIVE_HPP_SHA256="${TH20_ARCHIVE_HPP_SHA256}")
if(MSVC)
  target_compile_options(th20_source_archive_verify PRIVATE /W4 /permissive- /utf-8)
endif()
