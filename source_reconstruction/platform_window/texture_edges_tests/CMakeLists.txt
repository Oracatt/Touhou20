cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_texture_edges_oracle LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Original CPU comparison requires MSVC Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(th20_texture_edges_cpu_compare compare.cpp ../../sprite_renderer/texture_edges.cpp ../../../src/binary.cpp ../../../src/pe.cpp)
target_include_directories(th20_texture_edges_cpu_compare PRIVATE ../../../include ../../../native_recovered)
target_compile_features(th20_texture_edges_cpu_compare PRIVATE cxx_std_20)
target_compile_options(th20_texture_edges_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_link_options(th20_texture_edges_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO)
target_link_libraries(th20_texture_edges_cpu_compare PRIVATE bcrypt)
target_compile_definitions(th20_texture_edges_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused-test-fixture")
