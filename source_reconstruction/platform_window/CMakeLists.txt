cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_platform_window LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Original object layouts require MSVC -A Win32")
endif()
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT TARGET th20_runtime_core)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../runtime_core" "${CMAKE_CURRENT_BINARY_DIR}/runtime_core")
endif()
if(NOT TARGET th20_platform_services)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../platform_services" "${CMAKE_CURRENT_BINARY_DIR}/platform_services")
endif()
add_library(th20_platform_window STATIC layout.cpp window.cpp settings_dialog.cpp direct3d.cpp render_state.cpp input_queries.cpp global_state.cpp render_surfaces.cpp viewports.cpp directx_math.cpp shutdown.cpp select_viewport.cpp present.cpp surface_copy.cpp snapshot.cpp fonts.cpp frame_statistics.cpp graphics_callbacks.cpp scene_transitions.cpp)
target_compile_features(th20_platform_window PUBLIC cxx_std_17)
target_sources(th20_platform_window PRIVATE draw_controls.cpp)
target_compile_features(th20_platform_window PRIVATE cxx_std_20)
target_compile_options(th20_platform_window PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_include_directories(th20_platform_window PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ../core_scheduler)
target_include_directories(th20_platform_window PRIVATE ../../include ../../native_recovered)
target_link_libraries(th20_platform_window PUBLIC th20_runtime_core th20_platform_services user32 gdi32 d3d9 gdiplus winmm)
target_link_libraries(th20_platform_window PUBLIC th20_runtime_state th20_resource_manager)
# Resources must be linked into the final executable, not discarded inside a .lib.
target_sources(th20_platform_window INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/dialogs.rc")
option(TH20_BUILD_WINDOW_ORACLE "Build isolated original CPU layout/input query comparison" ON)
if(TH20_BUILD_WINDOW_ORACLE)
  add_executable(th20_window_cpu_compare cpu_compare.cpp layout.cpp input_queries.cpp viewports.cpp directx_math.cpp fonts.cpp select_viewport.cpp ../sprite_renderer/buffers.cpp ../../src/binary.cpp ../../src/pe.cpp)
  target_include_directories(th20_window_cpu_compare PRIVATE ../../include ../../native_recovered ../core_scheduler)
  target_compile_features(th20_window_cpu_compare PRIVATE cxx_std_17)
  target_compile_options(th20_window_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
  target_link_options(th20_window_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
  target_link_libraries(th20_window_cpu_compare PRIVATE th20_runtime_core bcrypt user32 gdi32)
endif()
