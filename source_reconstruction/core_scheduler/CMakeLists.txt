cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(th20_core_scheduler LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure this original-layout source module using MSVC -A Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(th20_core_scheduler STATIC scheduler.cpp)
target_compile_features(th20_core_scheduler PUBLIC cxx_std_17)
target_include_directories(th20_core_scheduler PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(th20_core_scheduler PRIVATE /W4 /utf-8)
option(TH20_BUILD_ORACLE "Build isolated original CPU differential test (requires local original EXE only when run)" ON)
if(TH20_BUILD_ORACLE)
  add_executable(th20_scheduler_cpu_compare cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
  target_include_directories(th20_scheduler_cpu_compare PRIVATE ../../native_recovered ../../include)
  target_compile_features(th20_scheduler_cpu_compare PRIVATE cxx_std_17)
  target_compile_options(th20_scheduler_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
  target_compile_definitions(th20_scheduler_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="not-used-by-scheduler")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/scheduler.cpp" SCHEDULER_CPP_SHA)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/scheduler.hpp" SCHEDULER_HPP_SHA)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS scheduler.cpp scheduler.hpp)
  target_compile_definitions(th20_scheduler_cpu_compare PRIVATE TH20_SCHEDULER_CPP_SHA="${SCHEDULER_CPP_SHA}" TH20_SCHEDULER_HPP_SHA="${SCHEDULER_HPP_SHA}")
  target_link_options(th20_scheduler_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
  target_link_libraries(th20_scheduler_cpu_compare PRIVATE th20_core_scheduler bcrypt)
endif()
