cmake_policy(SET CMP0091 NEW)
project(th20_title_oracle LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(th20_title_cpu_compare compare.cpp ../main_menu.cpp ../state.cpp ../background_math.cpp ../../pause_system/cursor_history.cpp ../../stone_menu/cursor.cpp ../../runtime_state/state.cpp ../../ecl_vm/vm.cpp ../../ecl_vm/math.cpp ../../core_scheduler/scheduler.cpp ../../runtime_core/runtime_core.cpp ../../../src/binary.cpp ../../../src/pe.cpp)
target_sources(th20_title_cpu_compare PRIVATE ../difficulty.cpp ../character.cpp ../loadout.cpp ../music.cpp ../stage_select.cpp ../replay_menu.cpp ../rank_entry.cpp ../practice.cpp ../../pause_system/menu_queries.cpp)
target_compile_features(th20_title_cpu_compare PRIVATE cxx_std_20)
target_sources(th20_title_cpu_compare PRIVATE ../replay_save.cpp)
target_sources(th20_title_cpu_compare PRIVATE ../player_data.cpp ../player_data_keys.cpp ../stones.cpp)
target_sources(th20_title_cpu_compare PRIVATE ../frame_core.cpp)
target_sources(th20_title_cpu_compare PRIVATE ../name_buffer.cpp)
target_sources(th20_title_cpu_compare PRIVATE ../lifecycle_core.cpp ../../runtime_core/worker.cpp)
target_compile_options(th20_title_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2 /Gy)
target_include_directories(th20_title_cpu_compare PRIVATE ../../../native_recovered ../../../include ../../core_scheduler)
target_link_options(th20_title_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /OPT:REF /STACK:8388608)
target_link_libraries(th20_title_cpu_compare PRIVATE bcrypt user32)
target_compile_definitions(th20_title_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused")
# Freeze the exact compilation inputs in the test binary. Changes trigger a
# configure and rebuild, so a later source edit cannot silently renew evidence.
get_target_property(title_inputs th20_title_cpu_compare SOURCES)
file(GLOB_RECURSE title_headers CONFIGURE_DEPENDS "../../*.hpp" "*.inl")
list(FILTER title_headers EXCLUDE REGEX "/build[^/]*/")
list(APPEND title_inputs ${title_headers} ../../../native_recovered/native_core.hpp ../../../tests/native_cpu_compare.cpp)
set(title_hashes "R\"json({)json\",")
set(title_comma "")
foreach(input IN LISTS title_inputs)
 get_filename_component(input "${input}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
 file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/../../.." "${input}")
 file(SHA256 "${input}" digest)
 string(APPEND title_hashes "\nR\"json(${title_comma}\"${relative}\":\"${digest}\")json\",")
 set(title_comma ",")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
endforeach()
string(APPEND title_hashes "\nR\"json(})json\"")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.hpp" "#pragma once\ninline constexpr const char* title_source_hashes[]={${title_hashes}};\n")
target_include_directories(th20_title_cpu_compare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
