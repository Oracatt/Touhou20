cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_title_draw_oracle LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(th20_title_draw_compare compare.cpp ../name_draw.cpp ../rank_draw.cpp ../replay_save_draw.cpp ../player_data_draw.cpp ../name_buffer.cpp ../../pause_system/text_style.cpp ../../pause_system/menu_queries.cpp ../../progress_state/profile_access.cpp ../../progress_state/records.cpp ../../game_session/session.cpp ../../replay_system/controller.cpp ../../text_renderer/format.cpp ../../runtime_core/runtime_core.cpp ../../core_scheduler/scheduler.cpp ../../../src/binary.cpp ../../../src/pe.cpp)
target_compile_features(th20_title_draw_compare PRIVATE cxx_std_20)
target_sources(th20_title_draw_compare PRIVATE ../../runtime_state/state.cpp ../../ecl_vm/vm.cpp ../../ecl_vm/math.cpp)
target_compile_options(th20_title_draw_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2 /Gy)
target_include_directories(th20_title_draw_compare PRIVATE ../../../native_recovered ../../../include ../../core_scheduler)
target_link_options(th20_title_draw_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /OPT:REF /STACK:8388608)
target_link_libraries(th20_title_draw_compare PRIVATE bcrypt user32)
target_compile_definitions(th20_title_draw_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused")
get_target_property(inputs th20_title_draw_compare SOURCES)
file(GLOB_RECURSE headers CONFIGURE_DEPENDS "../../*.hpp")
list(FILTER headers EXCLUDE REGEX "/build[^/]*/")
list(APPEND inputs ${headers} ../../../tests/native_cpu_compare.cpp ../../../native_recovered/native_core.hpp)
set(hashes "R\"json({)json\",")
set(comma "")
foreach(input IN LISTS inputs)
 get_filename_component(input "${input}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
 file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/../../.." "${input}")
 file(SHA256 "${input}" digest)
 string(APPEND hashes "\nR\"json(${comma}\"${relative}\":\"${digest}\")json\",")
 set(comma ",")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.hpp" "#pragma once\ninline constexpr const char* title_draw_hashes[]={${hashes}\nR\"json(})json\"};\n")
target_include_directories(th20_title_draw_compare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
