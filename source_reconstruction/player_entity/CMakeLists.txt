cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_player_entity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "PlayerInf requires MSVC -A Win32")
endif()
set(hash_header "#pragma once\ninline constexpr struct {const char* path;const char* sha;} source_hashes[]{\n")
foreach(path shot_callbacks.hpp shot_callbacks.cpp shot_callbacks_adapter.cpp shot_callbacks_cpu_cases.inc shot_geometry.hpp shot_geometry.cpp shot_geometry_cpu_cases.inc shot_hit.hpp shot_hit.cpp shot_hit_adapter.cpp shot_hit_cpu_cases.inc shot_hit_effect_cpu_cases.inc firing_at_position.hpp firing_at_position.cpp)
 file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" sha)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 string(APPEND hash_header "{\"${path}\",\"${sha}\"},\n")
endforeach()
foreach(path stage_reset.hpp stage_reset.cpp stage_reset_adapter.cpp stage_reset_cpu_cases.inc frame_helpers.hpp frame_helpers.cpp frame_helpers_adapter.cpp frame_helpers_cpu_cases.inc shot_controller_frame.hpp shot_controller_frame.cpp shot_controller_frame_adapter.cpp shot_controller_frame_cpu_cases.inc option_frame.hpp option_frame.cpp option_frame_adapter.cpp option_frame_cpu_cases.inc movement.hpp movement.cpp movement_adapter.cpp movement_cpu_cases.inc death_state.hpp death_state.cpp death_state_adapter.cpp death_state_cpu_cases.inc frame.hpp frame.cpp frame_adapter.cpp frame_cpu_cases.inc)
 file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" sha)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 string(APPEND hash_header "{\"${path}\",\"${sha}\"},\n")
endforeach()
foreach(path player.hpp collision.cpp collision_adapter.cpp events.hpp events.cpp events_adapter.cpp state_layout.hpp state_layout.cpp owner.hpp owner.cpp owner_adapter.cpp shots.hpp shots.cpp shots_adapter.cpp firing.hpp firing.cpp firing_adapter.cpp firing_cpu_cases.inc firing_native_services_cpu_cases.inc power.hpp power.cpp power_adapter.cpp shot_data.hpp shot_data.cpp shot_data_adapter.cpp initialize.hpp initialize.cpp initialize_adapter.cpp cpu_compare.cpp initialize_cpu_cases.inc ../archive/archive.cpp ../archive/archive.hpp ../sprite_renderer/animation.cpp ../damage_regions/geometry.cpp ../damage_regions/geometry.hpp ../runtime_state/motion.cpp ../runtime_state/motion.hpp ../ecl_vm/math.cpp ../ecl_vm/math.hpp)
 file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" sha)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 string(APPEND hash_header "{\"${path}\",\"${sha}\"},\n")
endforeach()
string(APPEND hash_header "};\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.hpp" CONTENT "${hash_header}")
add_executable(th20_player_entity_cpu_compare cpu_compare.cpp ../../src/binary.cpp ../../src/pe.cpp)
target_link_libraries(th20_player_entity_cpu_compare PRIVATE th20_player_entity bcrypt)
target_include_directories(th20_player_entity_cpu_compare PRIVATE ../../include ../../native_recovered "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(th20_player_entity_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_link_options(th20_player_entity_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /NXCOMPAT)
target_compile_definitions(th20_player_entity_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="not-used-by-player-entity")
if(NOT TARGET th20_ecl_vm)
 add_subdirectory(../ecl_vm player_ecl_dependency)
endif()
if(NOT TARGET th20_game_session)
 add_subdirectory(../game_session player_session_dependency)
endif()
if(NOT TARGET th20_runtime_core)
 add_subdirectory(../runtime_core player_runtime_dependency)
endif()
if(NOT TARGET th20_source_archive)
 add_subdirectory(../archive player_archive_dependency)
endif()
target_link_libraries(th20_player_entity_cpu_compare PRIVATE th20_source_archive)
add_library(th20_player_entity STATIC collision.cpp collision_adapter.cpp events.cpp events_adapter.cpp state_layout.cpp owner.cpp owner_adapter.cpp shots.cpp shots_adapter.cpp firing.cpp firing_adapter.cpp power.cpp power_adapter.cpp shot_data.cpp shot_data_adapter.cpp initialize.cpp initialize_adapter.cpp)
target_sources(th20_player_entity PRIVATE shot_callbacks.cpp shot_callbacks_adapter.cpp shot_geometry.cpp shot_hit.cpp shot_hit_adapter.cpp firing_at_position.cpp stage_reset.cpp stage_reset_adapter.cpp frame_helpers.cpp frame_helpers_adapter.cpp shot_controller_frame.cpp shot_controller_frame_adapter.cpp)
target_sources(th20_player_entity PRIVATE option_frame.cpp option_frame_adapter.cpp movement.cpp movement_adapter.cpp)
target_sources(th20_player_entity PRIVATE death_state.cpp death_state_adapter.cpp)
target_sources(th20_player_entity PRIVATE frame.cpp frame_adapter.cpp)
target_compile_features(th20_player_entity PUBLIC cxx_std_20)
target_compile_options(th20_player_entity PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_include_directories(th20_player_entity PRIVATE ../../native_recovered ../core_scheduler)
target_link_libraries(th20_player_entity PUBLIC th20_ecl_vm th20_game_session th20_runtime_core)
if(TARGET th20_runtime_state)
 target_link_libraries(th20_player_entity PUBLIC th20_runtime_state)
else()
 target_sources(th20_player_entity PRIVATE ../runtime_state/motion.cpp)
endif()
if(TARGET th20_resource_manager)
 target_link_libraries(th20_player_entity PUBLIC th20_resource_manager)
endif()
if(TARGET th20_sprite_renderer)
 target_link_libraries(th20_player_entity PUBLIC th20_sprite_renderer)
else()
 target_sources(th20_player_entity PRIVATE ../sprite_renderer/animation.cpp)
endif()
if(TARGET th20_damage_regions)
 target_link_libraries(th20_player_entity PUBLIC th20_damage_regions)
else()
 add_library(th20_player_collision_geometry STATIC ../damage_regions/geometry.cpp)
 target_compile_features(th20_player_collision_geometry PUBLIC cxx_std_20)
 target_compile_options(th20_player_collision_geometry PRIVATE /utf-8 /fp:strict /arch:SSE2)
 target_include_directories(th20_player_collision_geometry PRIVATE ../../native_recovered)
 target_link_libraries(th20_player_collision_geometry PUBLIC th20_ecl_vm)
 target_link_libraries(th20_player_entity PUBLIC th20_player_collision_geometry)
endif()
