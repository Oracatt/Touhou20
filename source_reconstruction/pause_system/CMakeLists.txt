cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_pause_system LANGUAGES CXX)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(th20_pause_system STATIC lifecycle.cpp owner_adapter.cpp transitions.cpp environment.cpp capture.cpp cursor_history.cpp ranking.cpp menu_queries.cpp resume.cpp menu.cpp)
target_compile_features(th20_pause_system PUBLIC cxx_std_20)
target_compile_options(th20_pause_system PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2)
target_include_directories(th20_pause_system PRIVATE ../../native_recovered ../../include ../core_scheduler)
target_link_libraries(th20_pause_system PUBLIC th20_gameplay th20_stone_menu th20_stage_completion)
target_sources(th20_pause_system PRIVATE draw.cpp draw_entries.cpp draw_ranking.cpp text_style.cpp)
