cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(th20_replay_oracle LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(th20_replay_cpu_compare compare.cpp ../records.cpp ../input.cpp ../../platform_services/configuration.cpp ../../game_session/session.cpp ../../core_scheduler/scheduler.cpp ../../../src/binary.cpp ../../../src/pe.cpp)
target_compile_features(th20_replay_cpu_compare PRIVATE cxx_std_20)
target_include_directories(th20_replay_cpu_compare PRIVATE ../../../native_recovered ../../../include ../../core_scheduler)
target_compile_options(th20_replay_cpu_compare PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2 /Gy)
target_link_options(th20_replay_cpu_compare PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /OPT:REF /STACK:8388608)
target_link_libraries(th20_replay_cpu_compare PRIVATE bcrypt user32)
target_compile_definitions(th20_replay_cpu_compare PRIVATE TH20_NATIVE_CORE_SHA256="unused")
set(REPLAY_HASH_INPUTS ../replay.hpp ../records.cpp ../input.hpp ../input.cpp ../../platform_services/configuration.hpp ../../platform_services/configuration.cpp ../../game_session/session.hpp ../../game_session/session.cpp ../../core_scheduler/scheduler.hpp ../../core_scheduler/scheduler.cpp)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.json" "{\n")
set(HASH_SEPARATOR "")
foreach(INPUT IN LISTS REPLAY_HASH_INPUTS)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${INPUT}" INPUT_HASH)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.json" "${HASH_SEPARATOR}  \"${INPUT}\":\"${INPUT_HASH}\"")
  set(HASH_SEPARATOR ",\n")
endforeach()
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/source_hashes.json" "\n}\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${REPLAY_HASH_INPUTS})
add_executable(th20_replay_demo_verify demo_verify.cpp ../load.cpp ../../archive/resource_manager.cpp ../../archive/archive.cpp ../../runtime_core/runtime_core.cpp ../../platform_services/services.cpp ../../platform_services/clock.cpp ../../game_session/session.cpp)
target_sources(th20_replay_demo_verify PRIVATE ../../core_scheduler/scheduler.cpp ../../platform_services/configuration.cpp)
target_compile_features(th20_replay_demo_verify PRIVATE cxx_std_20)
target_compile_options(th20_replay_demo_verify PRIVATE /W4 /utf-8 /fp:strict /arch:SSE2 /Gy)
target_include_directories(th20_replay_demo_verify PRIVATE ../../../native_recovered ../../../include ../../core_scheduler)
target_link_options(th20_replay_demo_verify PRIVATE /OPT:REF)
target_link_libraries(th20_replay_demo_verify PRIVATE user32 winmm)
